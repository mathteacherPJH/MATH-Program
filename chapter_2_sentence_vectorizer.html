from contextlib import asynccontextmanager from typing import List, Optional from fastapi import FastAPI, HTTPException from fastapi.middleware.cors import CORSMiddleware from kiwipiepy import Kiwi from pydantic import BaseModel # 글로벌 Kiwi 객체 선언 kiwi = None # 앱 시동/종료 시 실행될 Lifespan 이벤트 (Kiwi 사전 로드) @asynccontextmanager async def lifespan(app: FastAPI): global kiwi print("🚀 Kiwi 형태소 분석기 엔진을 메모리에 로드 중입니다...") kiwi = Kiwi() print("✅ Kiwi 엔진 준비 완료!") yield print("🛑 서버가 종료됩니다.") app = FastAPI(lifespan=lifespan) # 🌐 CORS 설정 (GitHub Pages 등 외부 클라이언트 통신 허용) app.add_middleware( CORSMiddleware, allow_origins=["*"], # 모든 Origin 허용 allow_credentials=True, allow_methods=["*"], # OPTIONS, POST 등 모든 HTTP 메서드 허용 allow_headers=["*"], # 모든 헤더 허용 ) # 요청 데이터 바디 정의 class AnalysisRequest(BaseModel): sentences: List[str] custom_vocab: Optional[List[str]] = [] # 헬스체크용 루트 엔드포인트 @app.get("/") def read_root(): return {"status": "ok", "message": "Kiwi 백엔드 엔진이 정상 작동 중입니다."} # 형태소 분석 및 벡터 변환 API 엔드포인트 @app.post("/api/analyze") def analyze_sentences(req: AnalysisRequest): global kiwi if not kiwi: raise HTTPException( status_code=500, detail="Kiwi 엔진이 아직 초기화되지 않았습니다." ) if not req.sentences: raise HTTPException( status_code=400, detail="분석할 문장이 제공되지 않았습니다." ) processed_corpus = [] extracted_tokens = set() # 1. 문장별 형태소 분석 (체분/용언 중심 추출) for sent in req.sentences: tokens = kiwi.tokenize(sent) # 명사(N), 동사(V), 형용사(VA), 어근(XR) 추출 filtered_words = [ t.form for t in tokens if t.tag.startswith("N") or t.tag.startswith("V") or t.tag.startswith("XR") ] processed_corpus.append(" ".join(filtered_words)) extracted_tokens.update(filtered_words) # 2. 단어 집합(Vocabulary) 생성 및 사용자 지정 단어 통합 vocabulary = sorted(list(extracted_tokens)) if req.custom_vocab: for cv in req.custom_vocab: if cv not in vocabulary: vocabulary.append(cv) # 3. 빈도수 및 원-핫 벡터 계산 frequency_matrix = [] one_hot_matrix = [] for corpus_str in processed_corpus: words = corpus_str.split() f_vec = [words.count(vocab_word) for vocab_word in vocabulary] oh_vec = [1 if count > 0 else 0 for count in f_vec] frequency_matrix.append(f_vec) one_hot_matrix.append(oh_vec) # 4. JSON 응답 반환 return { "vocabulary": vocabulary, "processed_corpus": processed_corpus, "one_hot": one_hot_matrix, "frequency": frequency_matrix, }
